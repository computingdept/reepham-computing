INDEPENDENT 1 · LIST

Create and edit a list

using System;
using System.Collections.Generic;

public class Program
{
    public static void Main()
    {
        List<string> names = new List<string> { "Amelia", "Ben", "Charlie", "Daisy" };
        names.Add("Ethan");
        names.Remove("Ben");

        foreach (string name in names)
        {
            Console.WriteLine(name);
        }
    }
}

Check: The list changes size and every remaining item is displayed.

INDEPENDENT 3 · LIST + SEARCH

Find an item

using System;
using System.Collections.Generic;

public class Program
{
    public static void Main()
    {
        List<string> courses = new List<string> { "Maths", "Physics", "Computer Science", "History", "Art" };
        Console.Write("Course: ");
        string wanted = Console.ReadLine();

        if (courses.Contains(wanted))
            Console.WriteLine("Found");
        else
            Console.WriteLine("Not found");
    }
}

Check: A present and absent course both produce sensible output.
